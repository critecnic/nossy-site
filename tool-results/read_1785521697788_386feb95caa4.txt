  1685→export async function GET(request: Request) {
  1686→  const { searchParams } = new URL(request.url);
  1687→  const country = searchParams.get("country");
  1688→  const type = searchParams.get("type");
  1689→  const sector = searchParams.get("sector");
  1690→  let filtered = [...jobs, ...companyJobs];
  1691→  if (country) {
  1692→    filtered = filtered.filter(j => j.country === country);
  1693→  }
  1694→  if (type && type !== "all") {
  1695→    filtered = filtered.filter(j => j.type.toLowerCase() === type.toLowerCase());
  1696→  }
  1697→  if (sector && sector !== "all") {
  1698→    filtered = filtered.filter(j => j.sector === sector);
  1699→  }
  1700→  return NextResponse.json(filtered);
  1701→}

  1703→// In-memory store for company-submitted jobs
  1704→const companyJobs: Job[] = [];
  1705→
  1706→export async function POST(request: Request) {
  1707→  try {
  1708→    const body = await request.json();
  1709→    const {
  1710→      companyName, companyUrl, companyEmail,
  1711→      title, location, country, countryName,
  1712→      salary, salaryMin, salaryMax, salaryCurrency, salaryPeriod,
  1713→      description, sector, type, remote
  1714→    } = body;
  1715→
  1716→    if (!companyName || !title || !location || !country || !description || !sector || !type) {
  1717→      return NextResponse.json({ error: "Missing required fields" }, { status: 400 });
  1718→    }
  1719→
  1720→    const newJob: Job = {
  1721→      id: jobs.length + companyJobs.length + 1,
  1722→      title,
  1723→      company: companyName,
  1724→      companyUrl: companyUrl || "",
  1725→      location,
  1726→      country,
  1727→      countryName: countryName || country,
  1728→      lat: 0, lng: 0,
  1729→      salary: salary || "Negotiable",
  1730→      salaryMin: salaryMin || 0,
  1731→      salaryMax: salaryMax || 0,
  1732→      salaryCurrency: salaryCurrency || "USD",
  1733→      salaryPeriod: salaryPeriod || "month",
  1734→      description,
  1735→      sector,
  1736→      type: type === "remote" ? "Remote" : type,
  1737→      contactEmail: companyEmail || "",
  1738→      paywall: false,
  1739→    };
  1740→
  1741→    companyJobs.push(newJob);
  1742→    return NextResponse.json({ success: true, job: newJob }, { status: 201 });
  1743→  } catch {
  1744→    return NextResponse.json({ error: "Invalid request" }, { status: 400 });
  1745→  }
