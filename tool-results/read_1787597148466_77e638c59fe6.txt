    61→function simpleHash(str: string): string {
    62→  let hash = 0;
    63→  for (let i = 0; i < str.length; i++) {
    64→    const char = str.charCodeAt(i);
    65→    hash = ((hash << 5) - hash) + char;
    66→    hash = hash & hash;
    67→  }

    98→function splitForTranslate(text: string, maxLen = 4000): string[] {

   103→    if (remaining.length <= maxLen) { chunks.push(remaining); break; }

   105→    if (idx < maxLen * 0.3) idx = -1;
   106→    if (idx === -1) { idx = remaining.lastIndexOf('\n', maxLen); if (idx < maxLen * 0.3) idx = -1; }
   107→    if (idx === -1) { idx = remaining.lastIndexOf('. ', maxLen); if (idx < maxLen * 0.3) idx = -1; }
   108→    if (idx === -1) { idx = remaining.lastIndexOf(' ', maxLen); if (idx < maxLen * 0.3) idx = -1; }

   110→    chunks.push(remaining.slice(0, idx));

   116→async function callGT(text: string, targetLang: string): Promise<string> {

   121→    const res = await fetch(url, { signal: controller.signal });
   122→    clearTimeout(timeout);
   123→    if (!res.ok) throw new Error('GT error: ' + res.status);
   124→    const data = await res.json();
   125→    if (!Array.isArray(data) || !Array.isArray(data[0])) throw new Error('Invalid GT response');

   127→    for (const seg of data[0]) {
   128→      if (Array.isArray(seg) && typeof seg[0] === 'string') result += seg[0];
   129→    }
   130→    return result || text;
   131→  } catch (err: any) {
   132→    clearTimeout(timeout);
   133→    if (err.name === 'AbortError') throw new Error('Translation timeout');
   134→    throw err;
   135→  }
   136→}

   138→export async function translateText(text: string, targetLang: Lang): Promise<string> {

   149→    }
   150→    const final = results.join('');
   151→    setCache(text, targetLang, final);
   152→    return final;
   153→  } catch {
   154→    return text;
   155→  }
   156→}
