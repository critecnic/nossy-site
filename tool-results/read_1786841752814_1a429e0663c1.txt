   390→def fmt_salary(val, symbol, period, code):
   391→    if period == "year":
   392→        if code == "JPY":

   398→def gen_jobs():
   399→    jobs = []
   400→    job_id = 1
   401→    random.seed(42)
   402→
   403→    for cc, cfg in COUNTRIES.items():
   404→        used = set()
   405→        sectors_shuffled = list(SECTORS)
   406→        random.shuffle(sectors_shuffled)
   407→
   408→        for sector in sectors_shuffled:
   409→            if sector not in cfg["companies"]:
   410→                continue
   411→            comps = cfg["companies"][sector]
   412→            comp = random.choice(comps)
   413→            title = random.choice(TITLES.get(sector, ["Analyst"]))
