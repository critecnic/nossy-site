  1890→const companyJobs: Job[] = [];
  1891→
  1892→export async function GET(request: Request) {
  1893→  const { searchParams } = new URL(request.url);
  1894→  const country = searchParams.get("country");
  1895→  const type = searchParams.get("type");
  1896→  const sector = searchParams.get("sector");
  1897→  let filtered = [...jobs, ...companyJobs];
  1898→  if (country) {
  1899→    filtered = filtered.filter(j => j.country === country);
  1900→  }
  1901→  if (type && type !== "all") {
  1902→    filtered = filtered.filter(j => j.type.toLowerCase() === type.toLowerCase());
  1903→  }
  1904→  if (sector && sector !== "all") {
  1905→    filtered = filtered.filter(j => j.sector === sector);
  1906→  }
  1907→  return NextResponse.json(filtered);
  1908→}
  1909→
  1910→export async function POST(request: Request) {
  1911→  try {
  1912→    const body = await request.json();
  1913→    const {
  1914→      companyName, companyUrl, companyEmail,
  1915→      title, location, country, countryName,
  1916→      salary, salaryMin, salaryMax, salaryCurrency, salaryPeriod,
  1917→      description, sector, type, remote
  1918→    } = body;
  1919→
  1920→    if (!companyName || !title || !location || !country || !description || !sector || !type) {
  1921→      return NextResponse.json({ error: "Missing required fields" }, { status: 400 });
  1922→    }
  1923→
  1924→    const newJob: Job = {
  1925→      id: jobs.length + companyJobs.length + 1,
  1926→      title,
  1927→      company: companyName,
  1928→      companyUrl: companyUrl || "",
  1929→      location,
  1930→      country,
  1931→      countryName: countryName || country,
  1932→      lat: 0, lng: 0,
  1933→      salary: salary || "Negotiable",
  1934→      salaryMin: salaryMin || 0,
  1935→      salaryMax: salaryMax || 0,
  1936→      salaryCurrency: salaryCurrency || "USD",
  1937→      salaryPeriod: salaryPeriod || "month",
  1938→      description,
  1939→      sector,
  1940→      type: type === "remote" ? "Remote" : type,
  1941→      contactEmail: companyEmail || "",
  1942→      paywall: false,
  1943→    };
  1944→
  1945→    companyJobs.push(newJob);
  1946→    return NextResponse.json({ success: true, job: newJob }, { status: 201 });
  1947→  } catch {
  1948→    return NextResponse.json({ error: "Invalid request" }, { status: 400 });
  1949→  }
