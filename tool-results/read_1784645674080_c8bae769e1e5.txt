   130→function AnimatedCounter({ end, label, suffix = "" }: { end: number; label: string; suffix?: string }) {
   131→  const [count, setCount] = useState(0);
   132→  const ref = useRef<HTMLDivElement>(null);
   133→  const started = useRef(false);
   134→  useEffect(() => {
   135→    const obs = new IntersectionObserver(([e]) => {
   136→      if (e.isIntersecting && !started.current) {
   137→        started.current = true;
   138→        const step = Math.max(1, Math.floor(end / 125));
   139→        let c = 0;
   140→        const t = setInterval(() => { c += step; if (c >= end) { setCount(end); clearInterval(t); } else setCount(c); }, 16);
   141→      }
   142→    }, { threshold: 0.5 });
   143→    if (ref.current) obs.observe(ref.current);
   144→    return () => obs.disconnect();
   145→  }, [end]);
   146→  return (<div ref={ref} className="text-center"><div className="text-3xl font-black text-white animate-count">{count.toLocaleString()}{suffix}</div><div className="mt-1 text-sm font-medium text-sky-200">{label}</div></div>);
   147→}

   160→function LanguageDropdown({ lang, setLang }: { lang: Lang; setLang: (l: Lang) => void }) {
   161→  const [open, setOpen] = useState(false);
   162→  const [search, setSearch] = useState("");
   163→  const ref = useRef<HTMLDivElement>(null);
   164→  const current = LANGUAGES.find((l) => l.code === lang)!;
   165→
   166→  useEffect(() => {
   167→    const handler = (e: MouseEvent) => { if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false); };
   168→    document.addEventListener("mousedown", handler);
   169→    return () => document.removeEventListener("mousedown", handler);
   170→  }, []);
   171→
   172→  const filtered = LANGUAGES.filter((l) => l.name.toLowerCase().includes(search.toLowerCase()) || l.code.toLowerCase().includes(search.toLowerCase()));
   173→
   174→  return (
   175→    <div ref={ref} className="relative">
   176→      <button onClick={() => { setOpen(!open); setSearch(""); }} className="flex items-center gap-1.5 rounded-lg border border-gray-200 px-2.5 py-1.5 text-sm transition-all hover:border-sky-300 hover:shadow-sm">
   177→        <span className="text-lg leading-none">{current.flag}</span>
   178→        <span className="hidden sm:inline text-xs font-semibold text-gray-600">{current.name}</span>
   179→        <svg className={`h-3 w-3 text-gray-400 transition-transform ${open ? "rotate-180" : ""}`} fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={2}><path strokeLinecap="round" strokeLinejoin="round" d="M19 9l-7 7-7-7" /></svg>
