    99→function AnimatedCounter({ end, label, suffix = "" }: { end: number; label: string; suffix?: string }) {
   100→  const [count, setCount] = useState(0);
   101→  const ref = useRef<HTMLDivElement>(null);
   102→  const started = useRef(false);
   103→  useEffect(() => {
   104→    const obs = new IntersectionObserver(([e]) => {
   105→      if (e.isIntersecting && !started.current) {
   106→        started.current = true;
   107→        const step = Math.max(1, Math.floor(end / 80));
   108→        let c = 0;
   109→        const t = setInterval(() => { c += step; if (c >= end) { setCount(end); clearInterval(t); } else setCount(c); }, 16);
   110→      }
   111→    }, { threshold: 0.3 });
   112→    if (ref.current) obs.observe(ref.current);
   113→    return () => obs.disconnect();
   114→  }, [end]);
   115→  return (<div ref={ref} className="text-center"><div className="text-3xl font-black text-white">{count.toLocaleString()}{suffix}</div><div className="mt-1 text-sm font-medium text-sky-200">{label}</div></div>);
   116→}

   118→function FilterChip({ label, onRemove }: { label: string; onRemove: () => void }) {
   119→  const [exiting, setExiting] = useState(false);
   120→  const handleRemove = () => { setExiting(true); setTimeout(onRemove, 280); };
   121→  if (exiting) return <span className="inline-flex items-center gap-1 rounded-full bg-sky-50 border border-sky-200 px-3 py-1 text-xs font-medium text-sky-700">{label}</span>;
   122→  return (
   123→    <span className="inline-flex items-center gap-1 rounded-full bg-sky-50 border border-sky-200 px-3 py-1 text-xs font-medium text-sky-700 transition-all hover:bg-sky-100">
   124→      {label}
   125→      <button onClick={handleRemove} className="ml-0.5 rounded-full p-0.5 transition-all hover:bg-sky-200 hover:rotate-90 duration-300">
   126→        <svg className="h-3 w-3" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}><path strokeLinecap="round" strokeLinejoin="round" d="M6 18L18 6M6 6l12 12" /></svg>
   127→      </button>
   128→    </span>
   129→  );
   130→}
